using System;

/// <summary>
/// 编辑器窗口：#SCRIPTNAME#
/// </summary>
/// <remarks>
/// 注意：该脚本需要放在 Editor 文件夹下
/// </remarks>
public class #SCRIPTNAME# : EditorWindow
{
    #region 启动
    /// <summary>
    /// 显示窗口
    /// </summary>
    [MenuItem("Tools/#SCRIPTNAME#")]
    public static void ShowWindow()
    {
        // 获取【窗口】
        #SCRIPTNAME# window = EditorWindow.GetWindow<#SCRIPTNAME#>(false, "#SCRIPTNAME#", true);

        // 设置【尺寸】
        float width  = 500.0f;
        float height = 800.0f;

        // 设置【尺寸下限】
        window.minSize = new Vector2(width, height);

        // 获取当前显示器的分辨率以设置窗口到屏幕中心
        window.position = new Rect
        (
            (Screen.currentResolution.width  - width ) * 0.5f,
            (Screen.currentResolution.height - height) * 0.5f,
            width,
            height
        );

        // 绘制窗口
        window.Show();

        // 聚焦窗口
        window.Focus();
    }
    #endregion

    #endregion

    }

    {
        #NOTRIM#
    }

#if UNITY_EDITOR
using System;
using UnityEditor;
using UnityEngine;

    #ROOTNAMESPACEBEGIN#
/// <summary>
/// 自定义属性绘制器：#目标类型class或结构体struct或特性attribute#
/// </summary>
/// <remarks>
/// <b>注意：</b>
/// <br/>
/// 该类型所在的脚本文件需要放置在 Editor 文件夹下；
/// </remarks>
[CustomPropertyDrawer(typeof(#目标类型class或结构体struct#))] // 关联脚本
public class #SCRIPTNAME# : PropertyDrawer
{
    #region 常量
    /// <summary>
    /// 缩进距离
    /// </summary>
    private const int INDENT_DISTANCE = 15;

    /// <summary>
    /// 空行高度
    /// </summary>
    private const int LINE_HIGHT = 6;
    #endregion

    #region 字段
    /// <summary>
    /// 缩进级别
    /// </summary>
    private int IndentLevel = 0;
    #endregion

    #region 属性
    /// <summary>
    /// 单行高度
    /// </summary>
    private float SingleLineHeight
    {
        get
        {
            return EditorGUIUtility.SingleLineHeight;
        }
    }

    /// <summary>
    /// 每行间距
    /// </summary>
    private float SpacePerLine
    {
        get
        {
            return EditorGUIUtility.SingleLineHeight + 2.0f;
        }
    }
    #endregion

    #region 生命周期方法
    /// <summary>
    /// 绘制 GUI 时
    /// </summary>
    /// <param name="position">指定需要绘制的 <paramref name="property"/> 的绘制区域。</param>
    /// <param name="property">指定需要绘制的序列化属性。</param>
    /// <param name="label">指定需要绘制的 <paramref name="property"/> 的 GUI 内容。</param>
    public override void OnGUI(Rect position, SerializedProperty property, GUIContent label)
    {
        // 重置缩进级别
        IndentLevel = 0;

        // 开始属性绘制
        EditorGUI.BeginProperty(position, label, property);

        // 绘制标题，高度为 1 行
        GUI.Label(new Rect(position.x, position.y, position.width, SingleLineHeight), label);

        // TODO: 绘制内容

        // 结束属性绘制
        EditorGUI.EndProperty();
    }

    /// <summary>
    /// 获取属性高度
    /// </summary>
    /// <param name="property">指定需要绘制的序列化属性。</param>
    /// <param name="label">指定需要绘制的 <paramref name="property"/> 的 GUI 内容。</param>
    /// <returns>返回值为属性在 <see cref="global::UnityEditor.InspectorWindow">检视窗口</see> 的绘制高度。</returns>
    public override float GetPropertyHeight(SerializedProperty property, GUIContent label)
    {
        // 计算整个 PropertyDrawer 需要的高度
        float result = 0;

        // 累加标题高度
        result += SpacePerLine;

        // TODO: 计算属性高度

        return result;
    }
    #endregion
}
#ROOTNAMESPACEEND#
#endif
